Downloads File Organizer

Sortiert Dateien im Download-Ordner automatisch anhand ihrer Endung in passende Unterordner wie Bilder, Dokumente oder Archive ein.

import java.io.File;
import java.nio.file.Files;
import java.nio.file.Path;

public class Main {
    public static void main(String[] args) throws Exception{


        String pfad = "C:/Users/patri/Downloads";
        File ordner = new File(pfad);
        File[] dateien = ordner.listFiles();

        for (File datei : dateien) {
            System.out.println(datei.getName());
            String zielOrdnerName;
            if (datei.isFile()) {
                String name = datei.getName().toLowerCase();
                if (name.endsWith(".jpg") || name.endsWith(".png") || name.endsWith(".jpeg")) {
                     zielOrdnerName = "Bilder";
                    System.out.println(name + " ist ein Bild");
                } else if (name.endsWith(".pdf") || name.endsWith(".txt")) {
                     zielOrdnerName = "Dokumente";
                    System.out.println(name + " ist ein Dokument");
                } else if (name.endsWith(".zip") || name.endsWith(".rar")) {
                     zielOrdnerName = "Archive";
                    System.out.println(name + " ist ein Archiv");
                } else {
                     zielOrdnerName = "Sonstiges";
                    System.out.println(name + " ist ein sonstiger Typ");
                }
                Path zielOrdner = Path.of(pfad, zielOrdnerName);
                Path zielDatei = Path.of(pfad, zielOrdnerName, datei.getName());

                Files.createDirectories(zielOrdner);
                Files.move(datei.toPath(), zielDatei);
                System.out.println(datei.getName() + " -> verschoben nach " + zielOrdnerName);

            }
        }
    }}

public class Held {

    String name;
    int lebenaktuell;
    int maxleben;
    int heiltrank;

    public Held(String name, int lebenaktuell, int maxleben, int heiltrank) {
        this.name = name;
        this.lebenaktuell = lebenaktuell;
        this.maxleben = maxleben;
        this.heiltrank = heiltrank;
    }

    public void nimmSchaden(int schaden) {
        this.lebenaktuell -= schaden;
        if (this.lebenaktuell < 0) {
            this.lebenaktuell = 0;
        }
        System.out.println(this.name + " erleidet " + schaden + " Schaden! Verbleibendes Leben: " + this.lebenaktuell + "/" + this.maxleben);
    }

    public void trinkeTrank() {
        if (this.heiltrank <= 0) {
            System.out.println("Fehler: Du hast keine Heiltränke mehr!");
            return;
        }

        this.lebenaktuell += 30;
        this.heiltrank--;


        if (this.lebenaktuell > this.maxleben) {
            this.lebenaktuell = this.maxleben;
        }

        System.out.println(this.name + " heilt sich! Aktuelles Leben: " + this.lebenaktuell + "/" + this.maxleben + " (Tränke übrig: " + this.heiltrank + ")");
    }

    public boolean lebtNoch() {
        return this.lebenaktuell > 0;
    }
}

import java.util.Random;
import java.util.Scanner;

public class Main {

    public static void main(String[] args) {
        Scanner scanner = new Scanner(System.in);
        Random wuerfel = new Random();

        Held meinHeld = new Held("Lanzelot", 100, 100, 2);
        Held gegner = new Held("Goblin", 80, 80, 0);

        System.out.println("Ein wilder " + gegner.name + " taucht auf!");

        while (meinHeld.lebtNoch() && gegner.lebtNoch()) {
            // Spieler-Aktion auswählen
            System.out.println("\n--- DU BIST AM ZUG ---");
            System.out.println("1: Angreifen");
            System.out.println("2: Heiltrank trinken");
            System.out.print("Deine Wahl: ");
            int wahl = scanner.nextInt();

            if (wahl == 1) {
                int heldschaden = wuerfel.nextInt(11) + 10;
                gegner.nimmSchaden(heldschaden);
            } else if (wahl == 2) {
                meinHeld.trinkeTrank();
            } else {
                System.out.println("Ungültige Eingabe! Du verpasst vor Verwirrung deinen Zug.");
            }

            // Prüfen, ob der Goblin besiegt wurde
            if (!gegner.lebtNoch()) {
                System.out.println("\nGlückwunsch! " + gegner.name + " wurde besiegt!");
                break;
            }

            // Gegenschlag des Goblins
            System.out.println("\n--- GEGNER IST AM ZUG ---");
            int gegnerschaden = wuerfel.nextInt(11) + 5;
            meinHeld.nimmSchaden(gegnerschaden);

            // Prüfen, ob Lanzelot besiegt wurde
            if (!meinHeld.lebtNoch()) {
                System.out.println("\nDu wurdest im Kampf besiegt... Game Over!");
                break;
            }
        }

        scanner.close();
    }
}
